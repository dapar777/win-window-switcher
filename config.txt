# Konfigurační soubor pro přepínání oken
# Formát: <zkratka> <regulární_výraz_titulku> <příkaz_pro_spuštění>
# Pokud zadáte zkratku a stisknete Enter:
# 1. Nejprve se vyhledá okno odpovídající regulárnímu výrazu.
# 2. Pokud se okno najde, aktivuje se (přenese do popředí).
# 3. Pokud se nenajde, spustí se zadaný příkaz.
#
# Příklad:
vn .*notes.* code -n c:/notes
gc .*chrome.* "C:\Program Files\Google\Chrome\Application\chrome.exe"

# Nastavení zobrazování bočního živého náhledu (true / false)
show_thumbnails false

# Nastavení zobrazování malých náhledů přímo v seznamu oken (true / false)
show_list_thumbnails true

# Měřítko zvětšení malých náhledů (např. 5 pro pětinásobné zvětšení z původních 48x30)
list_thumbnail_scale 5.0

# Výška okna přepínače
window_height 680

# Globální aktivační klávesová zkratka (např. win+tab, alt+caps, atd.)
# POZOR: Win+Tab je v systému Windows nativně rezervován pro Task View.
# Pokud registrace Win+Tab selže, přepínač automaticky použije bezpečný fallback.
# Doporučená kombinace pro stabilní fungování: alt+caps
hotkey_modifier alt
hotkey_key f12

# Chování při vzniku nového okna, pokud je aktivní skupina:
#   never              - nové okno se ignoruje
#   ask                - zobrazí dotaz; tlačítka: Ano / Ano dočasně / Vždy /
#                        Vždy do přepnutí / Ne - zůstat (skip, zůstat ve skupině) /
#                        Ne - opustit skupinu (skip + opustit skupinu) / Zavřít okno
#                        "Ano dočasně" = okno je ve skupině jen dokud je aktivní
#   always             - nové okno se vždy přidá bez ptaní
new_window_action ask

# Skrývat ikony na hlavním panelu pro okna mimo aktivní skupinu (true / false)
hide_taskbar_icons true

# Skrývat okna mimo aktivní skupinu i v přepínání Alt+Tab (true / false)
hide_alttab_icons true

# --- Automatická volba po vzniku nového okna (bez dotazu) ---
# Pro každou volbu jedna položka = regulární výraz. Hledá se v řetězci
# "<proces> <titulek>" (např. "chrome.exe Nova zprava - Gmail"), takže lze
# mířit na název aplikace i titulek. Auto-pravidla mají přednost před
# new_window_action ask/always.
#
# Vyhodnocují se v tomto pořadí (první shoda vyhrává):
#   1) new_window_auto_yes       - Ano: přidat do skupiny natrvalo
#   2) new_window_auto_yes_temp  - Ano dočasně: ve skupině jen dokud je aktivní
#   3) new_window_auto_no        - Ne - zůstat: vynechat, zůstat ve skupině
#   4) auto_close_windows        - Zavřít okno
#   5) new_window_auto_no_leave  - Ne - opustit skupinu: vynechat + opustit

# Příklad: new_window_auto_yes .*PL.SQL.*|.*Oracle.*
# new_window_auto_yes

# Příklad: new_window_auto_yes_temp .*Kalkulačka.*|.*Poznámkový.*
# new_window_auto_yes_temp

# Příklad: new_window_auto_no .*Outlook.*|.*Teams.*
# new_window_auto_no

# Příklad: auto_close_windows .*Calculator.*|.*Notepad.*
# auto_close_windows

# Příklad: new_window_auto_no_leave msedge.exe.*|.*Teams.*
# new_window_auto_no_leave
